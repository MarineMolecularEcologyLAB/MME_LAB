def create_network_file(corr_file, pval_file, output_prefix, corr_threshold=0.3, pval_threshold=0.01):
    corr = pd.read_csv(corr_file, sep='\t', index_col=0)
    pval = pd.read_csv(pval_file, sep='\t', index_col=0)
    edges = []
    for i in range(len(corr.index)):
        for j in range(i+1, len(corr.columns)):
            node1 = corr.index[i]
            node2 = corr.columns[j]
            cor_value = corr.iloc[i, j]
            p_value = pval.iloc[i, j]
            if abs(cor_value) >= corr_threshold and p_value < pval_threshold:
                interaction = 'positive' if cor_value > 0 else 'negative'
                edges.append({
                    'Source': node1,
                    'Target': node2,
                    'Correlation': cor_value,
                    'P-value': p_value,
                    'Interaction': interaction
                })
    edges_df = pd.DataFrame(edges)
    edges_df.to_csv(f'{output_prefix}_network.tsv', sep='\t', index=False)
    with open(f'{output_prefix}_network.sif', 'w') as f:
        for _, row in edges_df.iterrows():
            f.write(f"{row['Source']}\t{row['Interaction']}\t{row['Target']}\n")
    print(f"{output_prefix}: {len(edges_df)} edges written.")
    return edges_df

# 그룹별 전체 네트워크
create_network_file('cont_correlation.tsv', 'cont_pvalues.tsv', 'cont', corr_threshold=0.3, pval_threshold=0.01)

create_network_file('indirect_correlation.tsv', 'indirect_pvalues.tsv', 'indirect', corr_threshold=0.3, pval_threshold=0.01)
